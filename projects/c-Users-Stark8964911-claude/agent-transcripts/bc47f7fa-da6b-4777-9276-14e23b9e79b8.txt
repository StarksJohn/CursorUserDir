Merge img_163710.png into img_163721.png collage.
Creates a vertical stack: original 4-photo collage + the 5th photo.
"""
from pathlib import Path

try:
    from PIL import Image
except ImportError:
    print("Install Pillow: pip install Pillow")
    raise

COMMANDS_DIR = Path(__file__).parent
IMG_BASE = COMMANDS_DIR / "img_163721.png"
IMG_ADD = COMMANDS_DIR / "img_163710.png"
IMG_OUT = COMMANDS_DIR / "img_163721.png"  # Overwrite with merged result


def main():
    base = Image.open(IMG_BASE).convert("RGB")
    add = Image.open(IMG_ADD).convert("RGB")

    # Resize add image to match base width, preserve aspect ratio
    target_width = base.width
    ratio = target_width / add.width
    new_height = int(add.height * ratio)
    add_resized = add.resize((target_width, new_height), Image.Resampling.LANCZOS)

    # Create new canvas: base on top, add on bottom
    total_height = base.height + add_resized.height
    merged = Image.new("RGB", (target_width, total_height))
    merged.paste(base, (0, 0))
    merged.paste(add_resized, (0, base.height))

    merged.save(IMG_OUT, "PNG", optimize=True)
    print(f"Merged image saved to {IMG_OUT}")
    print(f"Dimensions: {target_width} x {total_height}")


if __name__ == "__main__":
    main()
