import { PrismaClient } from "@prisma/client";

const prisma = new PrismaClient();

async function main() {
  try {
    // Total registered users (exclude soft-deleted)
    const totalUsers = await prisma.user.count({
      where: { deletedAt: null },
    });

    // Active users (not deleted, optionally filtered by lastLoginAt)
    const activeUsers = await prisma.user.count({
      where: { deletedAt: null, isActive: true },
    });

    // Users by plan
    const usersByPlan = await prisma.user.groupBy({
      by: ["plan"],
      where: { deletedAt: null },
      _count: { id: true },
      orderBy: { _count: { id: true }: "desc" },
    });

    // Recent registrations (last 7 days)
    const sevenDaysAgo = new Date();
    sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 7);
    const recentRegistrations = await prisma.user.count({
      where: {
        deletedAt: null,
        createdAt: { gte: sevenDaysAgo },
      },
    });

    // Total contracts and analyses (for context)
    const totalContracts = await prisma.contract.count({
      where: { deletedAt: null },
    });
    const totalReports = await prisma.report.count();

    // Print results
    console.log("\n=== LeaseGuard User Statistics ===\n");
    console.log("Registration:");
    console.log(`  Total registered users: ${totalUsers}`);
    console.log(`  Active users:           ${activeUsers}`);
    console.log(`  New users (7 days):     ${recentRegistrations}`);
    console.log("");
    console.log("By Plan:");
    usersByPlan.forEach((row) => {
      console.log(`  ${row.plan.padEnd(10)}: ${row._count.id} users`);
    });
    console.log("");
    console.log("Usage:");
    console.log(`  Total contracts:  ${totalContracts}`);
    console.log(`  Total analyses:   ${totalReports}`);
    console.log("\n=== End ===\n");
  } catch (error) {
    console.error("Error fetching user stats:", error);
    if (error instanceof Error && error.message.includes("connect")) {
      console.error("\nTip: Ensure DATABASE_URL is set to your production Supabase connection string.");
      console.error("Get it from: Supabase Dashboard -> Settings -> Database -> Connection string");
    }
    process.exit(1);
  } finally {
    await prisma.$disconnect();
  }
}

main();
