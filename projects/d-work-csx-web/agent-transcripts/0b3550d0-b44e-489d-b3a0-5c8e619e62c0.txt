The CSX-web project has a **server-side printing architecture** with two parallel mechanisms:

### 二、两个方案的对比分析

#### 方案一: 取消打印服务 (纯浏览器方案)

**思路:** 移除后端打印机枚举和静默打印，改用浏览器原生 `window.print()` + PDF预览。

**改造内容:**
- 所有打印改为: 服务端生成PDF -> 前端打开PDF预览 -> 用户通过浏览器打印对话框手动打印
- 移除 `printServer.listPrinter()`、`PrintByHtml()`、`PrintByPdf()` 等方法
- 移除打印机关联设置页面

**优点:**
- 零部署成本，无需任何本地服务
- 实现最简单

**缺点:**
- **标签打印严重受损**: 药物标签、邮寄标签等高频批量打印场景，每次都需要手动选择打印机、确认打印，UX 极差

- **打印机预设丢失**: 无法为不同模板预设不同打印机

**结论: 不推荐作为完整方案。** 对标签打印场景影响过大。

#### 3. 打印服务封装 (`electron/print-service.js`)

```javascript
const { BrowserWindow } = require("electron");

const PrintService = {
  // Enumerate local printers
  getPrinters(mainWindow) {
    const printers = mainWindow.webContents.getPrinters();
    return printers.map((p, index) => ({
      printerId: index.toString(),
      printerMachineName: p.name,
      displayName: p.displayName || p.name,
      isDefault: p.isDefault,
      status: p.status,
    }));
  },

  // Silent print to a specific printer
  async printPDF(mainWindow, { printerName, pdfData, copies = 1, paperSize }) {
    return new Promise((resolve, reject) => {
      // Create a hidden window for printing
      const printWindow = new BrowserWindow({
        show: false,
        webPreferences: { contextIsolation: true },
      });

      // Load PDF data as data URL
      const base64 = Buffer.from(pdfData).toString("base64");
      printWindow.loadURL(`data:application/pdf;base64,${base64}`);

      printWindow.webContents.on("did-finish-load", () => {
        printWindow.webContents.print(
          {
            silent: true,
            deviceName: printerName,
            copies: copies,
            ...(paperSize && { pageSize: paperSize }),
          },
          (success, failureReason) => {
            printWindow.close();
            if (success) resolve(true);
            else reject(new Error(failureReason));
          }
        );
      });
    });
  },

  // Print HTML content to a specific printer
  async printHTML(mainWindow, { printerName, htmlContent, copies = 1, margins, paperSize }) {
    return new Promise((resolve, reject) => {
      const printWindow = new BrowserWindow({
        show: false,
        webPreferences: { contextIsolation: true },
      });

      printWindow.loadURL(
        `data:text/html;charset=utf-8,${encodeURIComponent(htmlContent)}`
      );

      printWindow.webContents.on("did-finish-load", () => {
        printWindow.webContents.print(
          {
            silent: true,
            deviceName: printerName,
            copies: copies,
            ...(margins && { margins }),
            ...(paperSize && { pageSize: paperSize }),
          },
          (success, failureReason) => {
            printWindow.close();
            if (success) resolve(true);
            else reject(new Error(failureReason));
          }
        );
      });
    });
  },
};

module.exports = PrintService;
```

#### 5. 修改 `printServer.js` (核心改动)

```javascript
import { Message } from "element-ui";
import { axios_post, axios_get } from "@/api/_axios";

// Detect if running in Electron
const isElectron = !!(window.csxPrintBridge && window.csxPrintBridge.isElectron);

const printServer = {
  // Get printer list: Electron native or backend API
  listPrinter: async (clinicId = "") => {
    if (isElectron) {
      // Use Electron native printer enumeration
      const printers = await window.csxPrintBridge.getPrinters();
      return printers.map((p) => ({
        printerId: p.printerMachineName, // Use printer name as ID for Electron
        printerMachineName: p.printerMachineName,
        displayName: p.displayName,
      }));
    }
    // Fallback: use backend API
    const rep = await axios_get("/api/Printer/ListPrinters", { clinicId });
    return rep.data.data.map((t) => ({
      printerId: t.id.toString(),
      printerMachineName: t.aliasName,
      displayName: t.displayName,
    }));
  },

  GetPrinter: async (printerId) => {
    if (isElectron) {
      const printers = await window.csxPrintBridge.getPrinters();
      const found = printers.find((p) => p.printerMachineName === printerId);
      return found
        ? { printerId: found.printerMachineName, printerMachineName: found.printerMachineName }
        : null;
    }
    const rep = await axios_get(`/api/Printer/GetPrinter/${printerId}`);
    return {
      printerId: rep.data.data.id.toString(),
      printerMachineName: rep.data.data.aliasName,
    };
  },

  // Print HTML content
  PrintByHtml: async ({ htmlContent, HaveSign, Top, Bottom, Left, Right, printerId, documentName }) => {
    if (isElectron) {
      await window.csxPrintBridge.printHTML({
        printerName: printerId, // In Electron mode, printerId is the printer name
        htmlContent,
        margins: {
          marginType: "custom",
          top: parseFloat(Top) || 0,
          bottom: parseFloat(Bottom) || 0,
          left: parseFloat(Left) || 0,
          right: parseFloat(Right) || 0,
        },
      });
      return true;
    }
    await axios_post("/api/Printer/HtmlToPdfPrint", {
      htmlContent, HaveSign, Top, Bottom, Left, Right,
      printerId, documentName,
    });
    return true;
  },

  // Print HTML content (with error handling)
  PrintByHtmlContent: async ({ htmlContent, HaveSign, Top, Bottom, Left, Right, printerId, documentName }) => {
    if (isElectron) {
      try {
        await window.csxPrintBridge.printHTML({
          printerName: printerId,
          htmlContent,
          margins: {
            marginType: "custom",
            top: parseFloat(Top) || 0,
            bottom: parseFloat(Bottom) || 0,
            left: parseFloat(Left) || 0,
            right: parseFloat(Right) || 0,
          },
        });
      } catch (err) {
        Message.closeAll();
        Message.error(err.message || "Print failed");
      }
      return true;
    }
    await axios_post("/api/Printer/HtmlContentToPdfPrint", {
      htmlContent, HaveSign, Top, Bottom, Left, Right,
      printerId, documentName,
    }).then((data) => {
      if (data.data.code !== 0 && data.data.message) {
        Message.closeAll();
        Message.error(data.data.message);
      }
    });
    return true;
  },

  // Print PDF file
  PrintByPdf: async ({ PdfFile, printerId }) => {
    if (isElectron) {
      // Convert ArrayBuffer/Blob to transferable format
      let pdfData = PdfFile;
      if (PdfFile instanceof Blob) {
        pdfData = await PdfFile.arrayBuffer();
      }
      await window.csxPrintBridge.printPDF({
        printerName: printerId,
        pdfData: Array.from(new Uint8Array(pdfData)),
      });
      return true;
    }
    const formData = new FormData();
    const blob = new Blob([PdfFile], { type: "application/pdf" });
    formData.append("PdfFile", blob);
    formData.append("printerId", printerId);
    await axios_post("/api/Printer/PdfPrint", formData);
    return true;
  },
};

export default printServer;
```

## 1. Background

### 1.1 Problem Statement

CSX-web is a clinic management system deployed as a web application. Currently, each clinic upgrade requires an additional step: deploying and configuring a separate **Print Service** on the local server/machine. This Print Service handles:

- Enumerating local printers (via backend API `/api/Printer/ListPrinters`)
- Converting HTML to PDF (via `/api/Printer/HtmlToPdfPrint`, `/api/Printer/HtmlContentToPdfPrint`)
- Sending print jobs to specific printers (via `/api/Printer/PdfPrint`)

This extra deployment step increases operational cost and complexity for every clinic upgrade.

```
Backend Server
```

### 2.1 Option A: Remove Print Service (Browser-Only)

Replace all server-side printing with browser `window.print()`.

| Aspect | Assessment |
|--------|-----------|
| Deployment | Zero local deployment needed |
| Label Printing | SEVERELY IMPACTED - each label requires manual browser print dialog |
| Silent Printing | NOT POSSIBLE - every print shows browser dialog |
| Paper Size Control | LIMITED - browser dialog may not handle label paper sizes |
| Printer Presets | LOST - cannot pre-assign printers to templates |
| Batch Printing | SEVERELY IMPACTED - each item requires manual confirmation |

**Verdict: NOT RECOMMENDED.** Unacceptable impact on label printing workflows.

### 2.2 Option B: Electron Wrapper (RECOMMENDED)

Wrap CSX-web in Electron, use native APIs for printer operations.

| Aspect | Assessment |
|--------|-----------|
| Deployment | Single `.exe` installer, no separate Print Service |
| Label Printing | ZERO IMPACT - full silent printing with custom paper sizes |
| Silent Printing | FULL SUPPORT - `webContents.print({silent: true, deviceName})` |
| Paper Size Control | FULL SUPPORT - `pageSize: {width, height}` in microns |
| Printer Presets | PRESERVED - printer list from `getPrintersAsync()` |
| Batch Printing | PRESERVED - same workflow, different print backend |

### 2.3 Option C: Tauri (Rust-based)

| Aspect | Assessment |
|--------|-----------|
| Bundle Size | Excellent (~3MB vs Electron's ~150MB) |
| Printer API | **NOT AVAILABLE** - no built-in printer enumeration |
| Implementation | Requires custom Rust code for printer operations |
| Learning Curve | Rust knowledge required |

**Verdict: NOT RECOMMENDED.** Critical gap in printer API.

### 2.4 Option D: .NET + WebView2

| Aspect | Assessment |
|--------|-----------|
| Printer API | Excellent (`System.Drawing.Printing`) |
| Platform | Windows-only |
| Runtime | Requires .NET runtime |
| Codebase | Separate C# project to maintain |
| Integration | WebView2 <-> C# IPC more complex than Electron IPC |

**Verdict: VIABLE but not optimal.** Adds C# maintenance burden.

### 2.5 Comparison Summary

| Criteria | Browser-Only | Electron | Tauri | .NET WebView2 |
|----------|-------------|----------|-------|---------------|
| Label Printing | BAD | GOOD | NEEDS WORK | GOOD |
| Silent Printing | NO | YES | NEEDS WORK | YES |
| Printer Enumeration | NO | YES (built-in) | NO (custom Rust) | YES (built-in) |
| Bundle Size | 0 | ~150MB | ~3MB | ~50MB |
| Code Reuse | 100% | 99% | 90% | 70% |
| Learning Curve | None | Low (JS) | High (Rust) | Medium (C#) |
| Auto-Update | N/A | Built-in | Built-in | Manual |
| Cross-Platform | Yes | Yes | Yes | Windows-only |

**WINNER: Electron** - Best balance of functionality, code reuse, and development effort.

## 3. Addressing Key Concerns

### 3.1 "Does Electron mean CSX becomes a desktop app?"

**Answer: Not exactly. It becomes a HYBRID.**

- The CSX-web Vue codebase remains a standard web application
- Electron provides an optional "shell" that loads the same web app
- Two deployment modes coexist:

| Mode | How it works | When to use |
|------|-------------|-------------|
| **Electron Mode** | Install CSX.exe, loads web app in Electron shell | Clinics that need printing |
| **Browser Mode** | Access via Chrome/Edge as before | Remote access, mobile, TV screens |

The Electron shell is an **additional deployment option**, not a replacement for the web version. The same Vue build output works in both modes.

### 3.2 Impact on Letter Printing

**Impact: ZERO**

Current flow:
```
HTML -> Backend API (HtmlToPdfForChromium) -> PDF -> printServer.PrintByPdf() -> Backend Print Service -> Printer
```

Electron flow:
```
HTML -> Backend API (HtmlToPdfForChromium) -> PDF -> Electron print API -> Printer (direct)
```

The only change is the **last mile**: instead of sending PDF to backend Print Service, Electron sends it directly to the local printer. The HTML generation, PDF conversion, template rendering, header/footer, logo, page numbers - all remain identical.

### 3.3 Impact on Label Printing

**Impact: ZERO**

Current flow:
```
HTML -> printServer.PrintByHtmlContent() -> Backend Print Service -> Printer
```

Electron flow:
```
HTML -> Electron hidden window -> webContents.print({silent, deviceName, pageSize}) -> Printer
```

Electron's `print()` API supports:
- Custom paper sizes: `pageSize: { width: 100000, height: 50000 }` (in microns)
- Silent printing: `silent: true`
- Specific printer: `deviceName: 'DYMO LabelWriter 450'`
- Copies: `copies: 3`
- Margins: `margins: { marginType: 'custom', top: 0, bottom: 0, left: 0, right: 0 }`

This is **functionally equivalent** to the current Print Service, with the advantage of running locally without a separate service.

### 3.4 Printer Enumeration

Current: `printServer.listPrinter()` -> Backend API -> Print Service enumerates printers

Electron: `webContents.getPrintersAsync()` -> Returns local printer list directly

Electron's `PrinterInfo` object provides:
```javascript
{
  name: "HP LaserJet MFP M132nw",
  displayName: "HP LaserJet MFP M132nw",
  description: "",
  status: 0,
  isDefault: true,
  options: { /* paper sizes, capabilities */ }
}
```

```javascript
const { app, BrowserWindow, ipcMain, session } = require("electron");
const path = require("path");
const { autoUpdater } = require("electron-updater");

let mainWindow;

function createWindow() {
  mainWindow = new BrowserWindow({
    width: 1440,
    height: 900,
    minWidth: 1024,
    minHeight: 768,
    webPreferences: {
      preload: path.join(__dirname, "preload.js"),
      contextIsolation: true,
      nodeIntegration: false,
      sandbox: false,
    },
    icon: path.join(__dirname, "../static/favicon.ico"),
    title: "CSX - Clinic Solution",
    autoHideMenuBar: true,
  });

  // Production: load built files; Development: load dev server
  if (process.env.ELECTRON_DEV === "true") {
    mainWindow.loadURL("http://localhost:8080");
    mainWindow.webContents.openDevTools();
  } else {
    mainWindow.loadFile(path.join(__dirname, "../dist/index.html"));
  }

  // Handle external links - open in system browser
  mainWindow.webContents.setWindowOpenHandler(({ url }) => {
    require("electron").shell.openExternal(url);
    return { action: "deny" };
  });

  mainWindow.on("closed", () => {
    mainWindow = null;
  });
}

// ============================================================
// IPC Handlers - Print Operations
// ============================================================

// Get list of local printers
ipcMain.handle("csx:getPrinters", async () => {
  if (!mainWindow) return [];
  try {
    const printers = await mainWindow.webContents.getPrintersAsync();
    return printers.map((p) => ({
      name: p.name,
      displayName: p.displayName || p.name,
      description: p.description,
      status: p.status,
      isDefault: p.isDefault,
    }));
  } catch (err) {
    console.error("Failed to get printers:", err);
    return [];
  }
});

// Silent print HTML content to a specific printer
ipcMain.handle("csx:printHTML", async (event, options) => {
  const {
    htmlContent,
    printerName,
    copies = 1,
    silent = true,
    margins,
    pageSize,
    landscape = false,
  } = options;

  return new Promise((resolve, reject) => {
    // Create a hidden window to render and print the HTML
    const printWin = new BrowserWindow({
      show: false,
      webPreferences: {
        contextIsolation: true,
        nodeIntegration: false,
      },
    });

    const fullHtml = `<!DOCTYPE html><html><head><meta charset="utf-8"></head><body>${htmlContent}</body></html>`;
    printWin.loadURL(
      `data:text/html;charset=utf-8,${encodeURIComponent(fullHtml)}`
    );

    printWin.webContents.on("did-finish-load", () => {
      const printOptions = {
        silent: silent,
        printBackground: true,
        copies: copies,
        landscape: landscape,
      };

      if (printerName) {
        printOptions.deviceName = printerName;
      }

      if (margins) {
        printOptions.margins = margins;
      }

      if (pageSize) {
        printOptions.pageSize = pageSize;
      }

      printWin.webContents.print(printOptions, (success, failureReason) => {
        printWin.close();
        if (success) {
          resolve({ success: true });
        } else {
          reject(new Error(failureReason || "Print failed"));
        }
      });
    });

    printWin.webContents.on(
      "did-fail-load",
      (event, errorCode, errorDescription) => {
        printWin.close();
        reject(new Error(`Failed to load print content: ${errorDescription}`));
      }
    );
  });
});

// Silent print PDF buffer to a specific printer
ipcMain.handle("csx:printPDF", async (event, options) => {
  const {
    pdfData, // Array of bytes
    printerName,
    copies = 1,
    silent = true,
    pageSize,
    landscape = false,
  } = options;

  return new Promise((resolve, reject) => {
    const printWin = new BrowserWindow({
      show: false,
      webPreferences: {
        contextIsolation: true,
        nodeIntegration: false,
        plugins: true, // Enable PDF plugin
      },
    });

    // Write PDF to a temp file and load it
    const fs = require("fs");
    const os = require("os");
    const tempPath = path.join(
      os.tmpdir(),
      `csx-print-${Date.now()}.pdf`
    );
    const buffer = Buffer.from(pdfData);
    fs.writeFileSync(tempPath, buffer);

    printWin.loadFile(tempPath);

    printWin.webContents.on("did-finish-load", () => {
      // Wait a bit for PDF to fully render
      setTimeout(() => {
        const printOptions = {
          silent: silent,
          printBackground: true,
          copies: copies,
          landscape: landscape,
        };

        if (printerName) {
          printOptions.deviceName = printerName;
        }

        if (pageSize) {
          printOptions.pageSize = pageSize;
        }

        printWin.webContents.print(printOptions, (success, failureReason) => {
          printWin.close();
          // Clean up temp file
          try {
            fs.unlinkSync(tempPath);
          } catch (e) {
            // ignore cleanup errors
          }
          if (success) {
            resolve({ success: true });
          } else {
            reject(new Error(failureReason || "PDF print failed"));
          }
        });
      }, 1000);
    });

    printWin.webContents.on(
      "did-fail-load",
      (event, errorCode, errorDescription) => {
        printWin.close();
        try {
          fs.unlinkSync(tempPath);
        } catch (e) {}
        reject(
          new Error(`Failed to load PDF for printing: ${errorDescription}`)
        );
      }
    );
  });
});

// ============================================================
// App Lifecycle
// ============================================================

app.whenReady().then(() => {
  createWindow();

  // Check for updates in production
  if (process.env.ELECTRON_DEV !== "true") {
    autoUpdater.checkForUpdatesAndNotify();
  }
});

app.on("window-all-closed", () => {
  if (process.platform !== "darwin") {
    app.quit();
  }
});

app.on("activate", () => {
  if (BrowserWindow.getAllWindows().length === 0) {
    createWindow();
  }
});
```

### 5.3 Modified src/utils/printServer.js

```javascript
import { Message } from "element-ui";
import { axios_post, axios_get } from "@/api/_axios";

/**
 * Detect if running inside Electron shell.
 * When in Electron, use native print APIs via csxPrintBridge.
 * When in browser, fall back to backend Print Service API.
 */
const isElectron = !!(
  window.csxPrintBridge && window.csxPrintBridge.isElectron
);

const printServer = {
  /**
   * Get list of available printers.
   * Electron: uses native printer enumeration.
   * Browser: calls backend API.
   * @param {string} clinicId - optional clinic ID filter
   * @returns {Promise<Array<{printerId, printerMachineName, displayName}>>}
   */
  listPrinter: async (clinicId = "") => {
    if (isElectron) {
      const printers = await window.csxPrintBridge.getPrinters();
      return printers.map((p) => ({
        printerId: p.name,
        printerMachineName: p.name,
        displayName: p.displayName || p.name,
      }));
    }
    // Fallback: backend API
    const rep = await axios_get("/api/Printer/ListPrinters", { clinicId });
    return rep.data.data.map((t) => ({
      printerId: t.id.toString(),
      printerMachineName: t.aliasName,
      displayName: t.displayName,
    }));
  },

  /**
   * Get a specific printer's info.
   * @param {string} printerId - printer ID or name
   * @returns {Promise<{printerId, printerMachineName}>}
   */
  GetPrinter: async (printerId) => {
    if (isElectron) {
      const printers = await window.csxPrintBridge.getPrinters();
      const found = printers.find((p) => p.name === printerId);
      if (found) {
        return {
          printerId: found.name,
          printerMachineName: found.name,
        };
      }
      return null;
    }
    const rep = await axios_get(`/api/Printer/GetPrinter/${printerId}`);
    return {
      printerId: rep.data.data.id.toString(),
      printerMachineName: rep.data.data.aliasName,
    };
  },

  /**
   * Print HTML content (used by document printing).
   * Electron: renders HTML in hidden window and prints natively.
   * Browser: sends to backend for PDF conversion and printing.
   */
  PrintByHtml: async ({
    htmlContent,
    HaveSign,
    Top,
    Bottom,
    Left,
    Right,
    printerId,
    documentName,
  }) => {
    if (isElectron) {
      try {
        await window.csxPrintBridge.printHTML({
          htmlContent,
          printerName: printerId,
          margins: {
            marginType: "custom",
            top: Math.round((parseFloat(Top) || 0) * 3.78),
            bottom: Math.round((parseFloat(Bottom) || 0) * 3.78),
            left: Math.round((parseFloat(Left) || 0) * 3.78),
            right: Math.round((parseFloat(Right) || 0) * 3.78),
          },
        });
      } catch (err) {
        Message.closeAll();
        Message.error(err.message || "Print failed");
      }
      return true;
    }
    await axios_post("/api/Printer/HtmlToPdfPrint", {
      htmlContent,
      HaveSign,
      Top,
      Bottom,
      Left,
      Right,
      printerId: printerId,
      documentName,
    });
    return true;
  },

  /**
   * Print HTML content with error handling (used by label printing).
   * Electron: renders HTML in hidden window and prints natively.
   * Browser: sends to backend for printing.
   */
  PrintByHtmlContent: async ({
    htmlContent,
    HaveSign,
    Top,
    Bottom,
    Left,
    Right,
    printerId,
    documentName,
  }) => {
    if (isElectron) {
      try {
        await window.csxPrintBridge.printHTML({
          htmlContent,
          printerName: printerId,
          margins: {
            marginType: "custom",
            top: Math.round((parseFloat(Top) || 0) * 3.78),
            bottom: Math.round((parseFloat(Bottom) || 0) * 3.78),
            left: Math.round((parseFloat(Left) || 0) * 3.78),
            right: Math.round((parseFloat(Right) || 0) * 3.78),
          },
        });
      } catch (err) {
        Message.closeAll();
        Message.error(err.message || "Print failed");
      }
      return true;
    }
    await axios_post("/api/Printer/HtmlContentToPdfPrint", {
      htmlContent,
      HaveSign,
      Top,
      Bottom,
      Left,
      Right,
      printerId: printerId,
      documentName,
    })
      .then((data) => {
        if (data.data.code !== 0 && data.data.message) {
          Message.closeAll();
          Message.error(data.data.message);
        }
      });
    return true;
  },

  /**
   * Print PDF file (used by document printing after PDF merge).
   * Electron: loads PDF in hidden window and prints natively.
   * Browser: uploads PDF to backend for printing.
   */
  PrintByPdf: async ({ PdfFile, printerId }) => {
    if (isElectron) {
      try {
        let pdfBytes;
        if (PdfFile instanceof Blob) {
          const arrayBuffer = await PdfFile.arrayBuffer();
          pdfBytes = Array.from(new Uint8Array(arrayBuffer));
        } else if (PdfFile instanceof ArrayBuffer) {
          pdfBytes = Array.from(new Uint8Array(PdfFile));
        } else if (PdfFile instanceof Uint8Array) {
          pdfBytes = Array.from(PdfFile);
        } else {
          pdfBytes = Array.from(new Uint8Array(PdfFile));
        }
        await window.csxPrintBridge.printPDF({
          pdfData: pdfBytes,
          printerName: printerId,
        });
      } catch (err) {
        Message.closeAll();
        Message.error(err.message || "PDF print failed");
      }
      return true;
    }
    const formData = new FormData();
    const blob = new Blob([PdfFile], { type: "application/pdf" });
    formData.append("PdfFile", blob);
    formData.append("printerId", printerId);
    await axios_post("/api/Printer/PdfPrint", formData);
    return true;
  },
};

export default printServer;
```

## 6. Migration Strategy

### 6.1 Dual-Mode Operation

The modified `printServer.js` supports three modes automatically:

```
Priority 1: Electron Mode (window.csxPrintBridge detected)
  -> Use native Electron print APIs
  -> No backend Print Service needed

Priority 2: Backend API Mode (no Electron, backend available)
  -> Use existing /api/Printer/* endpoints
  -> Requires backend Print Service (current behavior)

Priority 3: Browser Fallback (no Electron, no backend)
  -> PDF preview + browser print dialog
  -> Already implemented via exportCommon2()
```

### 6.2 Deployment Scenarios

| Scenario | Deployment | Print Method |
|----------|-----------|-------------|
| New clinic install | Install CSX.exe only | Electron native |
| Existing clinic upgrade | Install CSX.exe, keep Print Service temporarily | Either works |
| Remote access (browser) | No install needed | Backend API or browser print |
| TV screen / Queue display | Browser access | No printing needed |

### 6.3 Migration Steps

1. **Phase 1**: Deploy Electron version alongside existing web version
2. **Phase 2**: Clinics gradually switch to Electron version
3. **Phase 3**: Decommission backend Print Service after all clinics migrate
4. **Phase 4**: Remove Lodop/CLodop code (already mostly commented out)

| 9 | Auto-update integration | 1 day |
| 10 | Documentation & deployment guide | 0.5 day |
| **Total** | | **~12 days** |

---

## 10. Risk Mitigation

| Risk | Mitigation |
|------|-----------|
| Electron PDF rendering differs from backend | Keep backend PDF generation for complex documents; Electron only handles final print dispatch |
| Printer name mismatch between Electron and backend | Use printer name (not ID) as the common identifier |
| Label paper size not recognized | Use explicit `pageSize: { width, height }` in microns instead of named sizes |
| Electron update breaks printing | Pin Electron version; test before upgrading |
| Clinics resist installing desktop app | Keep browser mode as fallback; Electron is optional |

---

## 11. Conclusion

**Recommended Solution: Electron Wrapper**

- Eliminates the need for separate Print Service deployment
- Zero impact on letter and label printing functionality
- Minimal code changes (only `printServer.js` + 6 settings components)
- Dual-mode operation preserves browser access capability
- Single installer simplifies clinic upgrades
- Auto-update reduces future maintenance burden
- Industry-proven approach (VS Code, Slack, Discord all use Electron)

The Electron approach transforms the deployment from "web app + Print Service" to "single desktop app", while keeping the web version available for scenarios that don't require printing (TV screens, remote access, mobile).

## 2. 问题背景

### 2.1 现状

CSX-web 是一套面向诊所的管理系统。当前每家诊所部署升级时，除了更新前端静态文件外，还需要额外部署和配置一个**独立的 Print Service（打印服务）**。该服务运行在诊所本地机器上，负责:

- 枚举本地打印机资源
- 接收前端发来的 HTML/PDF 打印任务
- 将打印任务分派到指定打印机

### 2.2 痛点

- 每家诊所升级都需要人工部署配置 Print Service
- Print Service 需要与后端 API 配合，增加运维复杂度
- 不同诊所的网络环境不同，Print Service 的连接配置各异
- 升级时需要同时更新前端 + 后端 + Print Service 三个部分

### 2.3 需求目标

简化或取消 Print Service 的部署环节，具体有两个方向:
1. **方向一**: 取消打印服务，CSX 不再枚举本地打印机（需评估对信件、标签打印的影响）
2. **方向二**: 用 Electron 或 .NET UI 框架套壳 CSX 网页，在壳内实现本地打印机枚举（仍需安装 Windows 应用，但不再需要额外的 Print Service）

## 4. 方案对比与选型

### 4.1 方案一: 取消打印服务（纯浏览器方案）

**思路**: 移除后端打印机枚举和静默打印，所有打印改为浏览器 `window.print()` 对话框。

| 评估项 | 结果 |
|--------|------|
| 部署成本 | 零，无需任何本地服务 |
| 打印机枚举 | 不支持 -- 无法在 CSX 内选择打印机 |
| 静默打印 | 不支持 -- 每次打印都弹出浏览器对话框 |
| 标签打印 | **严重受损** -- 批量标签每张都需手动确认，纸张尺寸控制差 |
| 信件打印 | 中等影响 -- 每次需手动选择打印机 |
| 账单/报告 | 中等影响 -- 可接受但体验下降 |

**结论: 不推荐。** 对标签批量打印场景影响不可接受。

### 4.2 方案二: Electron 套壳（推荐）

**思路**: 用 Electron 包裹现有 CSX-web，利用 Electron 原生 API 实现打印机枚举和静默打印。

| 评估项 | 结果 |
|--------|------|
| 部署成本 | 安装一个 `.exe`，不需要额外 Print Service |
| 打印机枚举 | 完整支持 -- `webContents.getPrintersAsync()` 内置 API |
| 静默打印 | 完整支持 -- `webContents.print({silent: true, deviceName})` |
| 自定义纸张 | 完整支持 -- `pageSize: {width, height}` 微米级精度 |
| 标签打印 | **零影响** -- 与现有体验完全一致 |
| 信件打印 | **零影响** -- 与现有体验完全一致 |
| 代码改动 | 极小 -- 仅改 `printServer.js` 一个文件 + 新增 Electron 壳层 |
| 包体积 | ~150MB（含 Chromium 引擎） |

### 4.3 方案三: Tauri（Rust 框架）

**思路**: 用 Tauri 代替 Electron，利用系统 WebView 渲染网页。

| 评估项 | 结果 |
|--------|------|
| 包体积 | 优秀（~3MB） |
| 打印机枚举 | **不支持** -- Tauri 没有内置打印机枚举 API，需自写 Rust 代码 |
| 静默打印 | 有限支持 -- 需通过 Rust 后端实现 |
| 学习成本 | 高 -- 需要 Rust 开发能力 |

**结论: 不推荐。** 核心需求（打印机枚举）无内置支持，与项目目标矛盾。

### 4.4 方案四: .NET + WebView2

**思路**: 用 .NET WinForms/WPF + WebView2 控件承载 CSX 网页。

| 评估项 | 结果 |
|--------|------|
| 打印机枚举 | 完整支持 -- `System.Drawing.Printing.PrinterSettings.InstalledPrinters` |
| 静默打印 | 完整支持 -- .NET 打印 API |
| 平台限制 | 仅 Windows |
| 开发语言 | C# -- 需维护独立的 .NET 项目 |
| 代码复用 | ~70% -- WebView2 与 JS 的通信比 Electron IPC 更复杂 |

**结论: 可行但非最优。** 增加了 C# 维护成本。

### 4.5 选型总结

| 标准 | 纯浏览器 | Electron | Tauri | .NET WebView2 |
|------|---------|----------|-------|---------------|
| 标签打印影响 | 严重 | **零** | 需自开发 | **零** |
| 信件打印影响 | 中等 | **零** | 需自开发 | **零** |
| 打印机枚举 | 不支持 | **内置** | 无内置 | **内置** |
| 静默打印 | 不支持 | **内置** | 需自开发 | **内置** |
| 代码复用率 | 100% | **99%** | 90% | 70% |
| 学习成本 | 无 | **低 (JS)** | 高 (Rust) | 中 (C#) |
| 自动更新 | 不适用 | **内置** | 内置 | 需自建 |
| 包体积 | 0 | ~150MB | ~3MB | ~50MB |

**最终选择: Electron** -- 在打印功能完整性、代码复用率、开发效率三方面达到最佳平衡。

## 5. 关键问题解答

### 5.1 Electron 是否意味着项目变成了桌面应用?

**不完全是。项目变成"混合模式":**

CSX-web 的 Vue 代码仍然是标准 Web 应用，**不需要修改任何业务组件代码**。Electron 只是提供了一个可选的"桌面壳"来加载同一份 Web 构建产物。

两种部署模式并存:

| 模式 | 运行方式 | 适用场景 | 打印能力 |
|------|---------|---------|---------|
| **Electron 模式** | 安装 `CSX.exe`，在 Electron 窗口中运行 | 诊所前台（需要打印） | 完整（静默打印、打印机枚举） |
| **浏览器模式** | 通过 Chrome/Edge 访问 URL | 远程访问、TV 叫号屏、移动端 | 基础（浏览器打印对话框） |

本质上是给 Web 应用多了一个"桌面入口"，而非替代 Web 版本。构建产物（`dist/` 目录）完全相同，一份代码同时服务两种模式。

### 5.2 为什么 Electron 对打印影响最小?

因为改动仅在**打印的"最后一公里"**:

```
【不变】Vue 组件构建 HTML 内容
    |
    v
【不变】后端 API 生成 PDF（HtmlToPdfForChromium）
    |
    v
【不变】pdf-lib 合并 PDF、处理份数、添加页码
    |
    v
【仅此处改变】发送打印任务
  - 之前: printServer.PrintByPdf() -> 后端 API -> Print Service -> 打印机
  - 现在: Electron print API -> 直接发送到本地打印机
```

HTML 模板渲染、PDF 转换、页眉页脚、Logo、页码 -- 所有这些都完全不变。

### 5.3 打印机设置页面怎么办?

目前 6 个设置页面直接调用 Lodop 获取打印机列表。改造后统一使用 `printServer.listPrinter()`，该方法会自动检测运行环境:

```
Electron 环境 -> 调用 window.csxPrintBridge.getPrinters() -> 本地打印机列表
浏览器环境   -> 调用后端 /api/Printer/ListPrinters -> 后端返回打印机列表
```

```javascript
const { app, BrowserWindow, ipcMain, shell } = require("electron");
const path = require("path");
const fs = require("fs");
const os = require("os");

let mainWindow;

function createWindow() {
  mainWindow = new BrowserWindow({
    width: 1440,
    height: 900,
    minWidth: 1024,
    minHeight: 768,
    webPreferences: {
      preload: path.join(__dirname, "preload.js"),
      contextIsolation: true,  // Security: isolate renderer from Node.js
      nodeIntegration: false,  // Security: disable Node.js in renderer
      sandbox: false,
    },
    icon: path.join(__dirname, "../static/favicon.ico"),
    title: "CSX - Clinic Solution",
    autoHideMenuBar: true,
  });

  // Load the CSX-web app
  if (process.env.ELECTRON_DEV === "true") {
    // Development: load from Webpack dev server
    mainWindow.loadURL("http://localhost:80");
    mainWindow.webContents.openDevTools();
  } else {
    // Production: load built files
    mainWindow.loadFile(path.join(__dirname, "../dist/index.html"));
  }

  // Open external links in system browser
  mainWindow.webContents.setWindowOpenHandler(({ url }) => {
    if (url.startsWith("http")) {
      shell.openExternal(url);
      return { action: "deny" };
    }
    return { action: "allow" };
  });

  mainWindow.on("closed", () => {
    mainWindow = null;
  });
}

// ============================================================
// IPC Handler: Get local printer list
// ============================================================
ipcMain.handle("csx:getPrinters", async () => {
  if (!mainWindow) return [];
  try {
    const printers = await mainWindow.webContents.getPrintersAsync();
    return printers.map((p) => ({
      name: p.name,
      displayName: p.displayName || p.name,
      description: p.description,
      status: p.status,
      isDefault: p.isDefault,
    }));
  } catch (err) {
    console.error("[CSX] Failed to get printers:", err);
    return [];
  }
});

// ============================================================
// IPC Handler: Silent print HTML content
// ============================================================
ipcMain.handle("csx:printHTML", async (event, options) => {
  const {
    htmlContent,
    printerName,
    copies = 1,
    silent = true,
    margins,
    pageSize,
    landscape = false,
  } = options;

  return new Promise((resolve, reject) => {
    const printWin = new BrowserWindow({
      show: false,
      webPreferences: {
        contextIsolation: true,
        nodeIntegration: false,
      },
    });

    // Wrap HTML content in a full document
    const fullHtml = `<!DOCTYPE html>
      <html><head><meta charset="utf-8"></head>
      <body>${htmlContent}</body></html>`;
    printWin.loadURL(
      `data:text/html;charset=utf-8,${encodeURIComponent(fullHtml)}`
    );

    printWin.webContents.on("did-finish-load", () => {
      const printOptions = {
        silent,
        printBackground: true,
        copies,
        landscape,
      };
      if (printerName) printOptions.deviceName = printerName;
      if (margins) printOptions.margins = margins;
      if (pageSize) printOptions.pageSize = pageSize;

      printWin.webContents.print(printOptions, (success, failureReason) => {
        printWin.close();
        if (success) resolve({ success: true });
        else reject(new Error(failureReason || "Print failed"));
      });
    });

    printWin.webContents.on("did-fail-load", (ev, code, desc) => {
      printWin.close();
      reject(new Error(`Failed to load print content: ${desc}`));
    });
  });
});

// ============================================================
// IPC Handler: Silent print PDF buffer
// ============================================================
ipcMain.handle("csx:printPDF", async (event, options) => {
  const {
    pdfData,
    printerName,
    copies = 1,
    silent = true,
    pageSize,
    landscape = false,
  } = options;

  return new Promise((resolve, reject) => {
    const printWin = new BrowserWindow({
      show: false,
      webPreferences: {
        contextIsolation: true,
        nodeIntegration: false,
        plugins: true, // Enable PDF viewer plugin
      },
    });

    // Write PDF to temp file, then load it
    const tempPath = path.join(os.tmpdir(), `csx-print-${Date.now()}.pdf`);
    const buffer = Buffer.from(pdfData);
    fs.writeFileSync(tempPath, buffer);
    printWin.loadFile(tempPath);

    printWin.webContents.on("did-finish-load", () => {
      // Brief delay to allow PDF rendering
      setTimeout(() => {
        const printOptions = {
          silent,
          printBackground: true,
          copies,
          landscape,
        };
        if (printerName) printOptions.deviceName = printerName;
        if (pageSize) printOptions.pageSize = pageSize;

        printWin.webContents.print(printOptions, (success, failureReason) => {
          printWin.close();
          try { fs.unlinkSync(tempPath); } catch (e) { /* ignore */ }
          if (success) resolve({ success: true });
          else reject(new Error(failureReason || "PDF print failed"));
        });
      }, 1000);
    });

    printWin.webContents.on("did-fail-load", (ev, code, desc) => {
      printWin.close();
      try { fs.unlinkSync(tempPath); } catch (e) { /* ignore */ }
      reject(new Error(`Failed to load PDF: ${desc}`));
    });
  });
});

// ============================================================
// App lifecycle
// ============================================================
app.whenReady().then(() => {

    try {
      const { autoUpdater } = require("electron-updater");
      autoUpdater.checkForUpdatesAndNotify();
    } catch (e) {
      // electron-updater not available, skip
    }
  }
});

});

app.on("activate", () => {
```

### 7.3 src/utils/printServer.js -- 修改后的打印服务

```javascript
import { Message } from "element-ui";
import { axios_post, axios_get } from "@/api/_axios";

// Detect Electron environment
const isElectron = !!(
  window.csxPrintBridge && window.csxPrintBridge.isElectron
);

const printServer = {
  // Get printer list
  // Electron: native enumeration; Browser: backend API
  listPrinter: async (clinicId = "") => {
    if (isElectron) {
      const printers = await window.csxPrintBridge.getPrinters();
      return printers.map((p) => ({
        printerId: p.name,
        printerMachineName: p.name,
        displayName: p.displayName || p.name,
      }));
    }
    const rep = await axios_get("/api/Printer/ListPrinters", { clinicId });
    return rep.data.data.map((t) => ({
      printerId: t.id.toString(),
      printerMachineName: t.aliasName,
      displayName: t.displayName,
    }));
  },

  // Get specific printer info
  GetPrinter: async (printerId) => {
    if (isElectron) {
      const printers = await window.csxPrintBridge.getPrinters();
      const found = printers.find((p) => p.name === printerId);
      return found
        ? { printerId: found.name, printerMachineName: found.name }
        : null;
    }
    const rep = await axios_get(`/api/Printer/GetPrinter/${printerId}`);
    return {
      printerId: rep.data.data.id.toString(),
      printerMachineName: rep.data.data.aliasName,
    };
  },

  // Print by HTML (document printing)
  PrintByHtml: async ({
    htmlContent, HaveSign, Top, Bottom, Left, Right, printerId, documentName,
  }) => {
    if (isElectron) {
      try {
        await window.csxPrintBridge.printHTML({
          htmlContent,
          printerName: printerId,
          margins: {
            marginType: "custom",
            top: Math.round((parseFloat(Top) || 0) * 3.78),
            bottom: Math.round((parseFloat(Bottom) || 0) * 3.78),
            left: Math.round((parseFloat(Left) || 0) * 3.78),
            right: Math.round((parseFloat(Right) || 0) * 3.78),
          },
        });
      } catch (err) {
        Message.closeAll();
        Message.error(err.message || "Print failed");
      }
      return true;
    }
    await axios_post("/api/Printer/HtmlToPdfPrint", {
      htmlContent, HaveSign, Top, Bottom, Left, Right,
      printerId, documentName,
    });
    return true;
  },

  // Print by HTML content (label printing, with error handling)
  PrintByHtmlContent: async ({
    htmlContent, HaveSign, Top, Bottom, Left, Right, printerId, documentName,
  }) => {
    if (isElectron) {
      try {
        await window.csxPrintBridge.printHTML({
          htmlContent,
          printerName: printerId,
          margins: {
            marginType: "custom",
            top: Math.round((parseFloat(Top) || 0) * 3.78),
            bottom: Math.round((parseFloat(Bottom) || 0) * 3.78),
            left: Math.round((parseFloat(Left) || 0) * 3.78),
            right: Math.round((parseFloat(Right) || 0) * 3.78),
          },
        });
      } catch (err) {
        Message.closeAll();
        Message.error(err.message || "Print failed");
      }
      return true;
    }
    await axios_post("/api/Printer/HtmlContentToPdfPrint", {
      htmlContent, HaveSign, Top, Bottom, Left, Right,
      printerId, documentName,
    }).then((data) => {
      if (data.data.code !== 0 && data.data.message) {
        Message.closeAll();
        Message.error(data.data.message);
      }
    });
    return true;
  },

  // Print PDF file (after pdf-lib merge)
  PrintByPdf: async ({ PdfFile, printerId }) => {
    if (isElectron) {
      try {
        let pdfBytes;
        if (PdfFile instanceof Blob) {
          const ab = await PdfFile.arrayBuffer();
          pdfBytes = Array.from(new Uint8Array(ab));
        } else if (PdfFile instanceof ArrayBuffer) {
          pdfBytes = Array.from(new Uint8Array(PdfFile));
        } else if (PdfFile instanceof Uint8Array) {
          pdfBytes = Array.from(PdfFile);
        } else {
          pdfBytes = Array.from(new Uint8Array(PdfFile));
        }
        await window.csxPrintBridge.printPDF({
          pdfData: pdfBytes,
          printerName: printerId,
        });
      } catch (err) {
        Message.closeAll();
        Message.error(err.message || "PDF print failed");
      }
      return true;
    }
    const formData = new FormData();
    const blob = new Blob([PdfFile], { type: "application/pdf" });
    formData.append("PdfFile", blob);
    formData.append("printerId", printerId);
    await axios_post("/api/Printer/PdfPrint", formData);
    return true;
  },
};

export default printServer;
```

- 每家诊所升级都需要确保 Print Service（后端服务或 CLodop）在本地正常运行
- 后端打印 API 需要能够访问本地打印机，这要求后端服务部署在诊所本地或有本地代理
- 不同诊所的网络环境不同，Print Service 的连接配置各异
- 升级时可能需要同时更新前端 + 后端 + 本地打印服务三个部分
- CLodop 方案正在被淘汰，但后端方案仍然需要本地部署配置

### 2.4 需求目标

简化或取消 Print Service 的部署环节，具体有两个方向:
1. **方向一**: 取消打印服务，CSX 不再枚举本地打印机（需评估对信件、标签打印的影响）
2. **方向二**: 用 Electron 或 .NET UI 框架套壳 CSX 网页，在壳内实现本地打印机枚举（仍需安装 Windows 应用，但不再需要额外的 Print Service）
