 /**
  * Register push notification client with backend server
  * Supports both Firebase FCM and Huawei HMS Push
  * @param token - Push token (FCM or HMS)
  * @param pushType - Type of push service: 'fcm' or 'hms' (default: 'fcm')
  */
 export const register_client = async (token?: string, pushType: 'fcm' | 'hms' = 'fcm') => {
   try {
     const url = constant.API_BASE_URL + '/notification/v1/push/register-client';
     console.log('api register_client url=', url);
     logUtils.addLog('info', 'api register_client url=', '', url);
     if (!token) {
       // Only try to get FCM token if pushType is 'fcm'
       // HMS tokens should always be passed explicitly
       if (pushType === 'fcm') {
         // 当网络访问Google Firebase不畅时，messaging().getToken()会花费数十秒才继续执行，造成登录无反应的现象，影响用户体验
         // android和iOS都存在此现象
         // 添加超时机制
         const tokenPromise = messaging().getToken();
         const timeoutPromise = new Promise<string>((_, reject) => {
           setTimeout(() => reject(new Error('FCM token fetch timeout')), 2000); // 2秒超时
         });
 
         try {
           token = await Promise.race([tokenPromise, timeoutPromise]);
         } catch (error) {
           // 如果获取 token 失败，可以选择跳过或使用默认值
           return null; // 或者继续执行其他逻辑
         }
       } else {
         console.log('api register_client: HMS token must be provided explicitly');
         return null;
       }
     }
     const deviceId = await DeviceInfo.getUniqueId();
     const body = {
       clientId: 'cs-mobile',
       token,
       deviceId,
       pushType // Add pushType to distinguish FCM from HMS tokens
     };
     console.log('api register_client body=', body);
     logUtils.addLog('info', 'api register_client body=', '', body);
-    let response = await fetch(url, {
+    let response = await fetchData(url, {
       method: 'POST', // 指定请求方法
       headers: headers(),
       body: JSON.stringify(body),
     });
     console.log('api register_client response=', response);
     logUtils.addLog('info', 'api register_client response=', '', response);
     if (!response.ok) {
       return null;
     } else {
       let json = await response.json();
       console.log('api register_client json=', json);
       logUtils.addLog('info', 'api register_client json=', '', json);
       return json;
     }
   } catch (error) {
     console.error('api register_client error=', error);
     logUtils.addLog('info', 'api register_client error=', '', error);
     return null;
   }
 };

   try {
     const url = constant.API_BASE_URL + '/Bill/send-bill-to-pay';
     console.log('api send_bill_to_pay url=', url);
     const body = {appointmentId, diagnosisJsonAsString, currencyCode, expenseItems};
     console.log('api send_bill_to_pay body=', body);
-    let response = await fetch(url, {
+    let response = await fetchData(url, {
       method: 'POST', // 指定请求方法

       body: JSON.stringify(body),
     });
     console.log('send_bill_to_pay response=', response);
     if (!response.ok) {
       return null;
     }
     let json = await response.json();
     console.log('api send_bill_to_pay json=', json);
     if (
       json.code ===
       200 /*只有200才是發送賬單成功*/ /*|| json.status == 'WAIT_PAYMENT'Wait-payment是发送账单失败，不是成功，因为已经发送过了*/
     ) {
       return true;
     }
     if (json.status && json.status !== 'BAD_REQUST') {
       emitEvent(globalEventName.showCommonAlertModal, {
         commonAlertModalTitle: global.i18n.t('prompt'),
         commonAlertModalContent:
           json.status == 'WAIT_PAYMENT'
             ? global.i18n.t('Waiting_patient_payment')
             : global.i18n.t('Missing_diagnostic'),
         commonAlertModalCancelTxt: global.i18n.t('cancel'),
       });
     } else {
       if (json.message == 'The payment amount must be greater than 0.') {
         emitEvent(globalEventName.showCommonAlertModal, {
           commonAlertModalTitle: global.i18n.t('prompt'),
           commonAlertModalContent: global.i18n.t('The_payment_amount_must_be_greater_than_0'),
           commonAlertModalCancelTxt: global.i18n.t('cancel'),
         });
       } else {
         emitEvent(globalEventName.showCommonAlertModal, {
           commonAlertModalTitle: global.i18n.t('prompt'),
           commonAlertModalContent: json.message,
           commonAlertModalCancelTxt: global.i18n.t('cancel'),
         });
       }
     }
     return null;
   } catch (error) {
     console.error('api send_bill_to_pay error=', error);
     return null;
   }
