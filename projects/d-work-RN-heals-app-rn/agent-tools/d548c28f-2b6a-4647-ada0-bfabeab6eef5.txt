+import { useState, useEffect, useCallback } from 'react';
+import { useLoading } from "@components/loading-modal/loading-context";
+import { OrganizationItem } from "@api/api-types";
+import { PanelTypeItem } from "../screens/medical-service-screen/medical-service-screen.props";
+
+interface FilterOptionsState {
+  panelTypeList: PanelTypeItem[];
+  countryList: string[];
+  stateList: string[];
+  regionHierarchy: Record<string, string[]>;
+}
+
+export const useMedicalFilterData = (serviceType: 'Medical' | 'Rescue') => {
+
+  const [organizationList, setOrganizationList] = useState<OrganizationItem[]>([]);
+  const [isLoading, setIsLoading] = useState<boolean>(false);
+  const [isInitialLoad, setIsInitialLoad] = useState<boolean>(true);
+  const { setLoading } = useLoading();
+
+  const [optionsState, setOptionsState] = useState<FilterOptionsState>({
+    panelTypeList: [],
+    countryList: [],
+    stateList: [],
+    regionHierarchy: {},
+  });
+
+  const [selectedPannelType, setSelectedPannelType] = useState<string>('all_clinic');
+  const [selectedCountry, setSelectedCountry] = useState<string>('all_region');
+  const [selectedState, setSelectedState] = useState<string>('all_state');
+
+  useEffect(() => {
+    const initData = async () => {
+      try {
+        const providerOptionsRes = await $api.getProviderOptions();
+
+        const currentServiceType = providerOptionsRes.data.serviceTypes?.find(
+          (item: any) => item.code === serviceType
+        );
+        let panelTypes = currentServiceType ? currentServiceType.panelTypes : [];
+
+        const allOption: PanelTypeItem = {
+          code: 'all_clinic',
+          names: [
+            { Lang: 'en-US', Value: 'All Clinic' },
+            { Lang: 'zh-Hans', Value: '全部机构' },
+            { Lang: 'zh-Hant', Value: '全部機構' }
+          ]
+        } as PanelTypeItem;
+        panelTypes = [allOption, ...panelTypes];
+
+        let regionList = providerOptionsRes.data.region || [];
+        const allCountry: string = 'all_region';
+        regionList = [allCountry, ...regionList];
+
+        const hierarchy = providerOptionsRes.data.regionHierarchy || {};
+
+        setOptionsState(prev => ({
+          ...prev,
+          panelTypeList: panelTypes,
+          countryList: regionList,
+          regionHierarchy: hierarchy,
+          stateList: [],
+        }));
+      } catch (e) {
+        console.error(`Failed to init data for ${serviceType}:`, e);
+      }
+    };
+
+    initData();
+  }, [serviceType]);
+
+  useEffect(() => {
+    if (selectedCountry && optionsState.regionHierarchy) {
+      const newList = optionsState.regionHierarchy[selectedCountry] || [];
+      const formattedList = ['all_state', ...newList.filter(s => s !== 'all_state')];
+
+      setOptionsState(prev => ({
+        ...prev,
+        stateList: formattedList
+      }));
+      setSelectedState('all_state');
+    } else {
+      setOptionsState(prev => ({
+        ...prev,
+        stateList: []
+      }));
+      setSelectedState('all_state');
+    }
+  }, [selectedCountry, optionsState.regionHierarchy]);
+
+  const fetchData = useCallback(async () => {
+    setIsLoading(true);
+    isInitialLoad && setLoading(true);
+
+    try {
+      const response = await $api.getProviderClinics({
+        serviceType: serviceType,
+        panelType: selectedPannelType === 'all_clinic' ? '' : selectedPannelType,
+        country: selectedCountry === 'all_region' ? '' : selectedCountry,
+        state: selectedState === 'all_state' ? '' : selectedState,
+        keyword: '',
+      });
+      if (response.success) {
+        setOrganizationList(response.data || []);
+      } else {
+        setOrganizationList([]);
+      }
+    } catch (e) {
+      console.error(`Failed to fetch ${serviceType} list:`, e);
+      setOrganizationList([]);
+    } finally {
+      setIsLoading(false);
+      setLoading(false);
+      setIsInitialLoad(false);
+    }
+  }, [selectedPannelType, selectedCountry, selectedState, serviceType, optionsState.panelTypeList.length]);
+
+  useEffect(() => {
+    fetchData();
+  }, [fetchData]);
+
+  return {
+    organizationList,
+    panelTypeList: optionsState.panelTypeList,
+    countryList: optionsState.countryList,
+    stateList: optionsState.stateList,
+    isLoading,
+    isInitialLoad,
+
+    selectedPannelType,
+    selectedCountry,
+    selectedState,
+
+    setSelectedPannelType,
+    setSelectedCountry,
+    setSelectedState,
+
+    refresh: fetchData,
+  };
+};
