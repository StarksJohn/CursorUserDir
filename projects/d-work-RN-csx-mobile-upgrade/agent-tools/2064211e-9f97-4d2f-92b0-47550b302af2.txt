 const MedicationNotesCustomRow: React.FC<MedicationNotesCustomRowProps> = memo(({

                                                                                   onAddMedicationNote,

                                                                                 }: MedicationNotesCustomRowProps) => {
   const {uiTheme} = useThemeProvider();
   const [searchQuery, setSearchQuery] = useState<string>('');
   const emitEvent = useGlobalEventEmitter();
   useGlobalEventListener(
     AutoPositionedPopupEventNames.searchQueryChange,
     useCallback(({tag, searchQuery: newQuery}: { tag: string; searchQuery: string }): void => {
       console.log('MedicationNotesCustomRow searchQueryChange event received=', {tag, searchQuery: newQuery});
       if (tag === global.i18n.t('Medication_notes')) {
         setSearchQuery(newQuery);
       }
     }, [])
   );
   const handleAddPress = useCallback((): void => {
     console.log('MedicationNotesCustomRow handleAddPress=', {searchQuery, selectedMedicationNotes})
     if (searchQuery === '' && !selectedMedicationNotes) {
       return;
     }
     // Use searchQuery text if no item selected from dropdown
     const textToAdd: string | undefined = searchQuery || selectedMedicationNotes?.title;
     console.log('MedicationNotesCustomRow textToAdd=', textToAdd)
     if (textToAdd && medicationNotesModalList && medicationNotesModalList.indexOf(textToAdd) === -1) {
       onAddMedicationNote && onAddMedicationNote(textToAdd);
       setSearchQuery(''); // Clear local search query after adding
       clearSelectedItem && clearSelectedItem();
       emitEvent(AutoPositionedPopupEventNames.searchQueryChange, {
         tag: global.i18n.t('Medication_notes'),
         searchQuery: ''
       })
     }
   }, [searchQuery, selectedMedicationNotes, medicationNotesModalList, onAddMedicationNote, clearSelectedItem]);
   const disabled = useMemo(() => {
     return searchQuery === '' && !selectedMedicationNotes
   }, [searchQuery, selectedMedicationNotes]);
   return (
     <View style={styles.usageRow}>
       {children}
       <TouchableOpacity
         disabled={disabled}
         style={[
           styles.medicationNotesInputRowAddBt,
           disabled && {backgroundColor: uiTheme.colors.$disabledBtn}
         ]}
         onPress={handleAddPress}
       >
         <Text style={[
           styles.medicationNotesInputRowAddBtText,
           disabled && {color: uiTheme.colors.$disabledBtnText}
         ]}>
           + {global.i18n.t('join')}
         </Text>
       </TouchableOpacity>
     </View>
   );
