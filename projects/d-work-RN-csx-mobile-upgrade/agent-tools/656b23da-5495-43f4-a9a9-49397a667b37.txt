+// src/utils/logger.ts
+
+// 保存原始的 console 方法
+const originalConsole = {
+  log: console.log,
+  info: console.info,
+  warn: console.warn,
+  error: console.error
+};
+
+class Logger {
+  private static instance: Logger;
+  private enabled: boolean = true;
+
+  private constructor() { }
+
+  static getInstance(): Logger {
+    if (!Logger.instance) {
+      Logger.instance = new Logger();
+    }
+    return Logger.instance;
+  }
+
+  /**
+   * 启用或禁用日志输出
+   * @param enabled 是否启用日志
+   */
+  setEnabled(enabled: boolean): void {
+    this.enabled = enabled;
+  }
+
+  /**
+   * 输出调试日志
+   * @param message 消息
+   * @param optionalParams 可选参数
+   */
+  log(message?: any, ...optionalParams: any[]): void {
+    if (this.enabled) {
+      originalConsole.log(`[LOG] ${new Date().toISOString()}`, message, ...optionalParams);
+    }
+  }
+
+  /**
+   * 输出信息日志
+   * @param message 消息
+   * @param optionalParams 可选参数
+   */
+  info(message?: any, ...optionalParams: any[]): void {
+    if (this.enabled) {
+      originalConsole.info(`[INFO] ${new Date().toISOString()}`, message, ...optionalParams);
+    }
+  }
+
+  /**
+   * 输出警告日志
+   * @param message 消息
+   * @param optionalParams 可选参数
+   */
+  warn(message?: any, ...optionalParams: any[]): void {
+    if (this.enabled) {
+      originalConsole.warn(`[WARN] ${new Date().toISOString()}`, message, ...optionalParams);
+    }
+  }
+
+  /**
+   * 输出错误日志
+   * @param message 消息
+   * @param optionalParams 可选参数
+   */
+  error(message?: any, ...optionalParams: any[]): void {
+    if (this.enabled) {
+      originalConsole.error(`[ERROR] ${new Date().toISOString()}`, message, ...optionalParams);
+    }
+  }
+
+  /**
+   * 输出调试信息（仅在开发环境）
+   * @param message 消息
+   * @param optionalParams 可选参数
+   */
+  debug(message?: any, ...optionalParams: any[]): void {
+    if (this.enabled && __DEV__) {
+      originalConsole.log(`[DEBUG] ${new Date().toISOString()}`, message, ...optionalParams);
+    }
+  }
+}
+
+// 创建全局实例
+const logger = Logger.getInstance();
+
+// 禁用所有 console.log 输出（但不影响我们保存的原始方法）
+console.log = (): void => { };
+console.info = (): void => { };
+console.warn = (): void => { };
+console.error = (): void => { };
+
+export default logger;
