/**
 * Enhanced Error Boundary with silent auto-recovery.
 *
 * Strategy:
 * - When a render error is caught, report it to Sentry silently.
 * - Auto-recover (re-render children) up to MAX_AUTO_RETRY times within RETRY_COOLDOWN_MS.
 * - If the same error keeps recurring beyond the retry limit, the app stays on the
 *   last successful render (children) rather than showing a blocking error screen.
 * - In __DEV__ mode, show a debug fallback so developers can see the error.
 */
class SentryErrorBoundary extends Component<Props, State> {
  // Track auto-recovery attempts to prevent infinite render loops
  private retryCount: number = 0;
  // Timestamp of the first error in the current retry window
  private firstErrorTime: number = 0;

  constructor(props: Props) {
    super(props);
    this.state = { hasError: false };
  }

  static getDerivedStateFromError(error: Error): State {
    // Mark error state so componentDidCatch can handle recovery
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: React.ErrorInfo) {
    console.warn('SentryErrorBoundary caught an error (silent recovery):', error?.message);

    // Report to Sentry if enabled
    if (global.$sentryLog) {
      try {
        Sentry.withScope((scope) => {
          scope.setLevel('error');
          scope.setTag('error_boundary', 'silent_recovery');
          scope.setTag('retry_count', String(this.retryCount));
          scope.setContext('error_info', {
            componentStack: errorInfo?.componentStack,
            errorBoundary: 'SentryErrorBoundary',
            retryCount: this.retryCount,
          });

          Sentry.addBreadcrumb({
            message: 'Silent error boundary triggered',
            category: 'error',
            level: 'error',
            data: {
              errorMessage: error?.message,
              retryCount: this.retryCount,
            },
          });

          Sentry.captureException(error);
        });
      } catch (sentryError) {
        // Sentry itself should never crash the error boundary
        console.warn('SentryErrorBoundary: Failed to report to Sentry:', sentryError);
      }
    }

    // Call custom error handler if provided
    if (this.props.onError) {
      try {
        this.props.onError(error, errorInfo);
      } catch (handlerError) {
        console.warn('SentryErrorBoundary: Custom onError handler threw:', handlerError);
      }
    }

    // Auto-recovery logic: reset error state so children re-render
    const now = Date.now();

    // Reset retry counter if cooldown window has passed
    if (now - this.firstErrorTime > RETRY_COOLDOWN_MS) {
      this.retryCount = 0;
      this.firstErrorTime = now;
    }

    if (this.retryCount < MAX_AUTO_RETRY) {
      this.retryCount++;
      console.warn(
        `SentryErrorBoundary: Auto-recovering (attempt ${this.retryCount}/${MAX_AUTO_RETRY})`
      );
      // Schedule recovery on next tick to avoid synchronous re-render loop
      setTimeout(() => {
        this.setState({ hasError: false, error: undefined });
      }, 0);
    } else {
      // Exceeded retry limit within cooldown window.
      // In production: still recover silently after a short delay to let the
      // error-causing condition potentially resolve (e.g. async data loading).
      console.warn(
        'SentryErrorBoundary: Max retries reached. Will attempt recovery after delay.'
      );
      setTimeout(() => {
        // Reset the retry counter and try again
        this.retryCount = 0;
        this.firstErrorTime = Date.now();
        this.setState({ hasError: false, error: undefined });
      }, 2000);
    }
  }

  handleRetry = () => {
    this.retryCount = 0;
    this.firstErrorTime = 0;
    this.setState({ hasError: false, error: undefined });
  };

  handleRestart = () => {
    if (global.navigationRef) {
      try {
        this.retryCount = 0;
        this.firstErrorTime = 0;
        this.setState({ hasError: false, error: undefined });
        global.navigationRef.reset('WelcomePage', {});
      } catch (e) {
        console.warn('Failed to reset navigation:', e);
      }
    }
  };

  render() {
    if (this.state.hasError && this.state.error) {
      // In DEV mode, show debug fallback so developers can see errors
      if (__DEV__) {
        const FallbackComponent = this.props.fallbackComponent || DevErrorFallback;
        return (
          <FallbackComponent
            error={this.state.error}
            onRetry={this.handleRetry}
            onRestart={this.handleRestart}
          />
        );
      }

      // In production: render children as-is while auto-recovery is scheduled.
      // The componentDidCatch above has already scheduled a setState to clear
      // the error. In the brief moment before that fires, render children
      // to avoid showing any error UI to the user.
      return this.props.children;
    }

    return this.props.children;
  }
}
