 /**
 * Compute HTTP Signature for request authentication
 * 
 * @param array $headerHash Array of header values
 * @param array $headers Array of header names to include in signature
 * @return string The computed signature
 */
 private function computeHttpSignature($headerHash, $headers)
 {
 $signingBase = implode("\n", array_map(
 function($header) use ($headerHash) {
 return strtolower($header) . ": " . $headerHash[$header];
 },
 $headers
 ));

 $hash = hash_hmac('sha256', $signingBase, base64_decode($this->signingKey), true);
 $signature = base64_encode($hash);

 return sprintf(
 'keyId="%s",algorithm="hmac-sha256",headers="%s",signature="%s"',
 $this->signingKeyId,
 implode(' ', $headers),
 $signature
 );
 }

 /**
 * Generate signed headers for API request
 * 
 * @param string $method HTTP method
 * @param string $endpoint API endpoint
 * @param array|null $data Request data
 * @return array Array of headers
 */
 private function generateSignedHeaders($method, $endpoint, $data = null)
 {
 $targetUrl = '/' . ltrim($endpoint, '/');
 $method = strtolower($method);
 $utcNow = gmdate('Y-m-d\TH:i:s\Z');
 $traceId = "a7ca5bc3-3c7b-40e2-9c38-7aeb41eccab8";

 $digest = null;
 if ($data !== null) {
 $jsonData = json_encode($data);
 $sha256digest = hash('sha256', $jsonData, true);
 $digest = 'SHA-256=' . base64_encode($sha256digest);
 }

 $headerHash = [
 'Request-Date-Time' => $utcNow,
 'Api-Version' => $this->apiVersion,
 'Trace-Id' => $traceId,
 'Authorization' => 'Bearer ' . $this->getValidAccessToken(),
 '(request-target)' => $method . ' ' . $targetUrl
 ];

 if ($digest) {
 $headerHash['Digest'] = $digest;
 }

 $signatureHeaders = ['(request-target)', 'Api-Version', 'Request-Date-Time', 'Trace-Id', 'Authorization'];
 if ($digest) {
 $signatureHeaders[] = 'Digest';
 }

 $signature = $this->computeHttpSignature($headerHash, $signatureHeaders);

 $headers = [
 'Authorization: Bearer ' . $this->getValidAccessToken(),
 'Api-Version: ' . $this->apiVersion,
 'Request-Date-Time: ' . $utcNow,
 'Trace-Id: ' . $traceId,
 'Signature: ' . $signature,
 'Accept: application/json',
 'Content-Type: application/json'
 ];

 if ($digest) {
 $headers[] = 'Digest: ' . $digest;
 }

 return $headers;
 }

 /**
 * Send HTTP request to the API
 * 
 * @param string $method HTTP method
 * @param string $url Full URL
 * @param array|null $data Request data
 * @return array Response data
 * @throws Exception If request fails
 */
 private function sendRequest($method, $url, $data = null)
 {
 try {
 $curl = curl_init();
 $endpoint = parse_url($url, PHP_URL_PATH);
 $headers = $this->generateSignedHeaders($method, $endpoint, $data);

 $options = [
 CURLOPT_URL => $url,
 CURLOPT_RETURNTRANSFER => true,
 CURLOPT_ENCODING => '',
 CURLOPT_MAXREDIRS => 10,
 CURLOPT_TIMEOUT => $this->timeout,
 CURLOPT_FOLLOWLOCATION => true,
 CURLOPT_HTTP_VERSION => CURL_HTTP_VERSION_1_1,
 CURLOPT_CUSTOMREQUEST => $method,
 CURLOPT_HTTPHEADER => $headers,
 CURLOPT_SSL_VERIFYPEER => false,
 CURLOPT_SSL_VERIFYHOST => false
 ];

 if ($data !== null) {
 $options[CURLOPT_POSTFIELDS] = json_encode($data);
 }

 curl_setopt_array($curl, $options);

 $response = curl_exec($curl);
 $httpCode = curl_getinfo($curl, CURLINFO_HTTP_CODE);

 if ($response === false) {
 throw new Exception('Curl error: ' . curl_error($curl));
 }

 curl_close($curl);

 $responseData = json_decode($response, true);

 if (json_last_error() !== JSON_ERROR_NONE) {
 throw new Exception('JSON decode error: ' . json_last_error_msg());
 }

 if ($httpCode >= 400) {
 throw new Exception(
 'API request failed with status ' . $httpCode .
 ': ' . (isset($responseData['message']) ? $responseData['message'] : 'Unknown error')
 );
 }

 return $responseData;
 } catch (Exception $e) {
 throw new Exception('API request failed: ' . $e->getMessage());
 }
 }

 /**
 * Get OAuth2 access token
 * 
 * @return string Access token
 * @throws Exception If token request fails
 */
 private function getAccessToken()
 {
 try {
 $curl = curl_init();

 $headers = [
 'Content-Type: application/json',
 'Api-Version: ' . $this->apiVersion,
 ];

 $postFields = http_build_query([
 'grant_type' => 'client_credentials',
 'client_id' => $this->clientId,
 'client_secret' => $this->clientSecret
 ]);

 $options = [
 CURLOPT_URL => $this->baseUrl . '/oauth2/token',
 CURLOPT_RETURNTRANSFER => true,
 CURLOPT_ENCODING => '',
 CURLOPT_MAXREDIRS => 10,
 CURLOPT_TIMEOUT => $this->timeout,
 CURLOPT_FOLLOWLOCATION => true,
 CURLOPT_HTTP_VERSION => CURL_HTTP_VERSION_1_1,
 CURLOPT_CUSTOMREQUEST => 'POST',
 CURLOPT_POSTFIELDS => $postFields,
 CURLOPT_HTTPHEADER => $headers,
 CURLOPT_SSL_VERIFYPEER => false,
 CURLOPT_SSL_VERIFYHOST => false
 ];

 curl_setopt_array($curl, $options);

 $response = curl_exec($curl);
 $httpCode = curl_getinfo($curl, CURLINFO_HTTP_CODE);

 if ($response === false) {
 throw new Exception('Curl error: ' . curl_error($curl));
 }

 curl_close($curl);

 $responseData = json_decode($response, true);

 if (json_last_error() !== JSON_ERROR_NONE) {
 throw new Exception('JSON decode error: ' . json_last_error_msg());
 }

 if ($httpCode >= 400) {
 throw new Exception(
 'Token request failed with status ' . $httpCode .
 ': ' . (isset($responseData['message']) ? $responseData['message'] : 'Unknown error')
 );
 }

 $this->accessToken = $responseData['accessToken'];
 $this->tokenExpiry = strtotime($responseData['expiresOn']);

 return $this->accessToken;
 } catch (Exception $e) {
 throw new Exception('Token request failed: ' . $e->getMessage());
 }
 }

 /**
 * Check if current token is expired or about to expire
 * 
 * @return bool True if token is expired or about to expire
 */
 private function isTokenExpired()
 {
 if (!$this->accessToken || !$this->tokenExpiry) {
 return true;
 }
 return $this->tokenExpiry <= (time() + 300); // 5 minutes buffer
 }

 /**
 * Get valid access token, refreshing if necessary
 * 
 * @return string Valid access token
 */
 private function getValidAccessToken()
 {
 if ($this->isTokenExpired()) {
 return $this->getAccessToken();
 }
 return $this->accessToken;
 }

 /**
 * Build signing base string from headers
 * 
 * @param array $headers Request headers
 * @param string $headerList Comma-separated list of headers to include
 * @return string Signing base string
 */
 private function buildSigningBase($headers, $headerList)
 {
 $headerNames = explode(' ', $headerList);
 $signingBase = [];

 foreach ($headerNames as $headerName) {
 $headerName = strtolower($headerName);
 
 if ($headerName === '(request-target)') {
 // Handle special (request-target) header
 $method = isset($_SERVER['REQUEST_METHOD']) ? strtolower($_SERVER['REQUEST_METHOD']) : 'post';
 $path = isset($_SERVER['REQUEST_URI']) ? $_SERVER['REQUEST_URI'] : '/';
 $signingBase[] = "(request-target): " . $method . " " . $path;
 } else {
 // Convert header name to proper format (e.g., 'api-version' to 'Api-Version')
 $headerKey = implode('-', array_map('ucfirst', explode('-', $headerName)));
 if (!isset($headers[$headerKey])) {
 throw new Exception("Required header not found: " . $headerKey);
 }
 $signingBase[] = $headerName . ": " . $headers[$headerKey];
 }
 }

 return implode("\n", $signingBase);
 }

 /**
 * Verify the signature of an incoming webhook request
 * 
 * @param array $headers Request headers
 * @param string $body Request body
 * @return bool True if signature is valid
 * @throws Exception If verification fails
 */
 public function verifyWebhookSignature($headers, $body)
 {
 $signatureHeader = isset($headers['Signature']) ? $headers['Signature'] : null;
 if (!$signatureHeader) {
 throw new Exception("Missing Signature header");
 }

 // Parse signature components
 $components = $this->parseSignatureHeader($signatureHeader);
 if (!$components) {
 throw new Exception("Invalid signature header format");
 }

 // Verify keyId matches
 if ($components['keyId'] !== $this->signingKeyId) {
 throw new Exception("Invalid keyId in signature");
 }

 // Verify algorithm
 if ($components['algorithm'] !== 'hmac-sha256') {
 throw new Exception("Invalid signature algorithm: " . $components['algorithm']);
 }

 // Build signing base from headers
 $signingBase = $this->buildSigningBase($headers, $components['headers']);

 // Compute expected signature
 $expectedSignature = $this->computeSignature($signingBase);

 // Compare signatures
 return hash_equals($expectedSignature, $components['signature']);
 }

 /**
 * Parse signature header into components
 * 
 * @param string $signatureHeader Raw signature header
 * @return array|null Array of signature components or null if invalid
 */
 private function parseSignatureHeader($signatureHeader)
 {
 $components = [];
 $parts = explode(',', $signatureHeader);
 foreach ($parts as $part) {
 $keyValue = explode('=', $part, 2);
 if (count($keyValue) !== 2) {
 return null;
 }

 $key = trim($keyValue[0]);
 $value = trim($keyValue[1], '"');

 switch ($key) {
 case 'keyId':
 case 'algorithm':
 case 'headers':
 case 'signature':
 $components[$key] = $value;
 break;
 default:
 return null;
 }
 }

 // Verify all required components are present
 $required = ['keyId', 'algorithm', 'headers', 'signature'];
 foreach ($required as $component) {
 if (!isset($components[$component])) {
 return null;
 }
 }

 return $components;
 }

 /**
 * Compute signature for a given signing base
 * 
 * @param string $signingBase The signing base string
 * @return string Base64 encoded signature
 */
 private function computeSignature($signingBase)
 {
 $hash = hash_hmac('sha256', $signingBase, base64_decode($this->signingKey), true);
 return base64_encode($hash);
 }

 /**
 * Verify the signature of an API response
 * 
 * @param array $headers Response headers
 * @param string $body Response body
 * @return bool True if signature is valid
 */
 public function verifyResponseSignature($headers, $body)
 {
 return $this->verifyWebhookSignature($headers, $body);
 }
}

 /**
 * Create a new payment request
 * 
 * @param float $amount The total amount to charge
 * @param string $orderId Unique order identifier
 * @param string $description Payment description
 * @param array $items Array of items in the payment
 * @return array Response containing payment details or error information
 */
 public function createPayment($amount, $orderId, $description, $items = [])
 {
 try {
 $shoppingCart = $this->formatShoppingCart($amount, $description, $items);

 $paymentData = [
 "totalAmount" => $amount,
 "currencyCode" => "HKD",
 "effectiveDuration" => 600, // 10 minutes
 "notificationUri" => PAYME_NOTIFICATION_URI,
 "appSuccessCallback" => PAYME_SUCCESS_CALLBACK,
 "appFailCallback" => PAYME_FAIL_CALLBACK,
 "merchantData" => [
 "orderId" => $orderId,
 "orderDescription" => $description,
 "additionalData" => "Order Reference: " . $orderId,
 "shoppingCart" => $shoppingCart
 ]
 ];

 $response = $this->paymeClient->post('/payments/paymentrequests', $paymentData);

 return [
 'success' => true,
 'paymentRequestId' => isset($response['paymentRequestId']) ? $response['paymentRequestId'] : null,
 'totalAmount' => isset($response['totalAmount']) ? $response['totalAmount'] : null,
 'currencyCode' => isset($response['currencyCode']) ? $response['currencyCode'] : null,
 'notificationUri' => isset($response['notificationUri']) ? $response['notificationUri'] : null,

 'appFailCallback' => isset($response['appFailCallback']) ? $response['appFailCallback'] : null,
 'createdTime' => isset($response['createdTime']) ? $response['createdTime'] : null,
 'effectiveDuration' => isset($response['effectiveDuration']) ? $response['effectiveDuration'] : null,
 'webLink' => isset($response['webLink']) ? $response['webLink'] : null,
 'appLink' => isset($response['appLink']) ? $response['appLink'] : null,
 'statusDescription' => isset($response['statusDescription']) ? $response['statusDescription'] : null,

 }
