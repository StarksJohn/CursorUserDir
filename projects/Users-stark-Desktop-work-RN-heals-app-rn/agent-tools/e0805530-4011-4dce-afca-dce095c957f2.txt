+  TouchableOpacity,
+  View,
+} from 'react-native';
+import {
+  AdvancedFlatList,
+  type AdvancedFlatListRef,
+  type FetchDataParams,
+  type ListData,
+  type RenderItemParams,
+} from 'react-native-advanced-flatlist';
+
+import type { Ticket, Claim, Preapproval, DocumentFileItem, Prescription } from '@api/api-types';
+import NavigationBar from "@components/navigation-bar/navigation-bar.tsx";
+import PageRightBt from "@components/page-right-bt/page-right-bt.tsx";
+import AppText from '@components/app-text/app-text';
+import DoctorAvatar from '@components/doctor-avatar/doctor-avatar';
+import constants from '@constants';
+import {
+  getTicketDisplayDateTime,
+  getTicketDoctorTypeLabel,
+  shouldCountReceipt,
+} from '@helper/ticket-display-helper';
+import {
+  SVGConsultCalendarIcon,
+  SVGConsultClockIcon,
+  SVGTabConsultIcon,
+  SVGChevronRightComponent,
+} from '../../assets/svg';
+import { ThemeContext } from '@style/theme/theme-context';
+import type { Theme } from '../../types/Theme';
+import styles, {
+  getMedicalRecordsScreenThemeStyles,
+} from './medical-records-list-screen.styles';
+import DashLine from '@components/dashLine/dashLine';
+
+interface TicketDateTime {
+  date: string;
+  time: string;
+}
+
+interface MedicalListScreenContentProps {
+  theme: Theme;
+}
+
+interface DocNumberAndTypes {
+  count: number;
+  types: string;
+}
+
+const CONSULT_PAGE_SIZE = 10;
+const MEDICAL_LIST_TAG = 'medical-records';
+
+const fetchTicketGroup = async (
+  pageIndex: number,
+  pageSize: number
+): Promise<Ticket[]> => {
+  try {
+    const tickets = await $api.medicalTicketList({
+      page: pageIndex,
+      limit: pageSize,
+      hasDocument: true,
+      statuses: ['completed'],
+    });
+
+    return Array.isArray(tickets) ? tickets : [];
+  } catch {
+    return [];
+  }
+};
+
+const getDisplayTimeZone = (): string | undefined =>
+  constants.$fake ? 'Asia/Shanghai' : undefined;
+
+const getTicketDateTime = (ticket: Ticket): TicketDateTime =>
+  getTicketDisplayDateTime(
+    ticket,
+    constants.i18n.locale,
+    getDisplayTimeZone(),
+    'iso'
+  );
+const getDocNumberAndTypes = (ticket: Ticket): DocNumberAndTypes => {
+  const documents = ticket?.consultation?.documents;
+  const includeReceipt = shouldCountReceipt(ticket);
+
+  type SourceItem = Claim | Preapproval | DocumentFileItem | Prescription;
+  type Source = { name: string; arr: SourceItem[] | null | undefined };
+
+  const sources: Source[] = [
+    { name: $i18n.t('claims'), arr: ticket?.claims },
+    { name: $i18n.t('preapprovals'), arr: ticket?.preapprovals },
+    {
+      name: $i18n.t('receipt'),
+      arr: includeReceipt ? documents?.receipts : [],
+    },
+    { name: $i18n.t('medicalCertificate'), arr: documents?.medicalCertificates },
+    { name: $i18n.t('prescription'), arr: ticket?.consultation?.prescriptions },
+    { name: $i18n.t('referralLetter'), arr: documents?.referralLetters },
+  ];
+
+  const nonEmpty = sources.filter(
+    ({ arr }) => Array.isArray(arr) && arr.length > 0
+  );
+
+  return {
+    count: nonEmpty.length,
+    types: nonEmpty.map(({ name }) => name).join(','),
+  };
+}
+const getDoctorAvatarId = (ticket: Ticket): string => {
+  const doctorId = ticket.doctor?.id;
+
+  return typeof doctorId === 'string' ? doctorId.trim() : '';
+};
+
+const MedicalRecordsListContent = ({
+  theme,
+}: MedicalListScreenContentProps): React.ReactElement => {
+  const listRef = useRef<AdvancedFlatListRef<Ticket> | null>(null);
+  const themeStyles = useMemo(
+    () => getMedicalRecordsScreenThemeStyles(theme),
+    [theme]
+  );
+
+  const handleOpenMedicalRecord = useCallback((ticket: Ticket): void => {
+    $NavigationActions.navigate($routeNames.medicalRecordsDetailScreen, {
+      ticketId: ticket.id,
+    });
+  }, []);
+
+  const fetchAppointments = useCallback(
+    async ({
+      pageIndex,
+      pageSize,
+    }: FetchDataParams<Ticket>): Promise<ListData<Ticket>> => {
+
+      const tickets = await fetchTicketGroup(
+        pageIndex,
+        pageSize
+      );
+
+      return {
+        items: tickets,
+        pageIndex: pageIndex + 1,
+        needLoadMore: tickets.length === pageSize,
+      };
+    },
+    []
+  );
+
+  const renderMedicalItem = useCallback(
+    ({ item }: RenderItemParams<Ticket>): React.ReactElement => {
+      const dateTime = getTicketDateTime(item);
+      const doctorName = item.doctor?.name ?? constants.i18n.t('no_data');
+      const specialty = getTicketDoctorTypeLabel(item, (key: string): string =>
+        constants.i18n.t(key)
+      );
+
+      const { count, types } = getDocNumberAndTypes(item);
+
+      const doctorAvatarId = getDoctorAvatarId(item);
+      const defaultDoctorAvatar = (
+        <SVGTabConsultIcon
+          color="#287A65"
+          height={20}
+          width={20}
+        />
+      );
+
+      return (
+        <TouchableOpacity
+          activeOpacity={0.8}
+          onPress={(): void => handleOpenMedicalRecord(item)}
+          style={[styles.card, themeStyles.card]}
+        >
+          <View style={styles.cardHeader}>
+            <View style={styles.doctorIdentity}>
+              <View
+                style={styles.doctorIconContainer}
+              >
+                {doctorAvatarId ? (
+                  <DoctorAvatar
+                    fallback={defaultDoctorAvatar}
+                    id={doctorAvatarId}
+                    style={styles.doctorAvatar}
+                  />
+                ) : (
+                  defaultDoctorAvatar
+                )}
+              </View>
+              <View style={styles.doctorText}>
+                <AppText
+                  ellipsizeMode="tail"
+                  numberOfLines={1}
+                  style={[styles.doctorName, themeStyles.primaryText]}
+                >
+                  {doctorName}
+                </AppText>
+                <AppText
+                  ellipsizeMode="tail"
+                  numberOfLines={1}
+                  style={[styles.doctorSpecialty, themeStyles.secondaryText]}
+                >
+                  {specialty}
+                </AppText>
+                <View style={styles.dateTimeRow}>
+                  <View style={styles.dateTimeItem}>
+                    <SVGConsultCalendarIcon
+                      color="#99A1AF"
+                      height={14}
+                      style={styles.dateTimeIcon}
+                      width={14}
+                    />
+                    <AppText
+                      style={[styles.dateTimeText, themeStyles.secondaryText]}
+                    >
+                      {dateTime.date}
+                    </AppText>
+                  </View>
+                  <View style={[styles.dateTimeItem, styles.dateTimeItemGap]}>
+                    <SVGConsultClockIcon
+                      color="#99A1AF"
+                      height={14}
+                      style={styles.dateTimeIcon}
+                      width={14}
+                    />
+                    <AppText
+                      style={[styles.dateTimeText, themeStyles.secondaryText]}
+                    >
+                      {dateTime.time}
+                    </AppText>
+                  </View>
+                </View>
+              </View>
+              <SVGChevronRightComponent fill={'#A0ACAB'} width={16} height={16} />
+            </View>
+          </View>
+
+          <DashLine style={styles.dashLine} lineWidth={1} />
+
+          <View style={styles.cardFooter}>
+            <View style={styles.docNumberView}>
+              <AppText
+                style={styles.docNumberText}
+              >
+                {count} {constants.i18n.t('documents')}
+              </AppText>
+            </View>
+            <AppText
+              style={styles.docTypeText}
+              numberOfLines={1}
+            >
+              {types}
+            </AppText>
+          </View>
+        </TouchableOpacity>
+      );
+    },
+    [handleOpenMedicalRecord, themeStyles]
+  );
+
+  const emptyComponent = useMemo(
+    (): React.ReactElement => (
+      <View style={styles.emptyState}>
+        <AppText style={[styles.emptyText, themeStyles.emptyText]}>
+          {constants.i18n.t('no_medical_record')}
+        </AppText>
+      </View>
+    ),
+    [themeStyles.emptyText]
+  );
+
+  return (
+    <View
+      style={[
+        theme.uiTheme?.commonStyles.pageContainer,
+        styles.page,
+        themeStyles.page,
+      ]}
+    >
+      <NavigationBar title={$i18n.t('my_medical_records')} onLeftPress={() => {
+        $NavigationActions.navigate($routeNames.mainTabNavigator)
+      }}
+        rightBt={<PageRightBt tag={$routeNames.medicationListScreen} />}
+      />
+      <AdvancedFlatList<Ticket>
+        ref={listRef}
+        autoRefresh
+        contentContainerStyle={styles.listContent}
+        fetchData={fetchAppointments}
+        initPageIndex={1}
+        keyExtractor={(item: Ticket): string => item.id}
+        ListEmptyComponent={emptyComponent}
+        pageSize={CONSULT_PAGE_SIZE}
+        renderItem={renderMedicalItem}
+        showsVerticalScrollIndicator
+        style={styles.list}
+        tag={MEDICAL_LIST_TAG}
+      />
+    </View>
+  );
+};
+
+const MedicalRecordsListScreen = (): React.ReactElement => {
+  return (
+    <ThemeContext.Consumer>
+      {(theme: Theme) => <MedicalRecordsListContent theme={theme} />}
+    </ThemeContext.Consumer>
+  );
+};
+
+export default MedicalRecordsListScreen;
